    cogant.runtime

    The cogant.runtime package provides the Active Inference agent runtime — the perception-action loop that consumes synthesized matrices (or any object exposing A, B, C, D) and executes belief updates, action selection, and multi-episode learning.

    Package

    cogant.runtime

    Active Inference agent runtime for COGANT synthesized packages.

    This module provides an executable perception-action loop that operates on the A/B/C/D matrices produced by :func:cogant.reverse.synthesizer.synthesize_package. The categorical arithmetic is pure Python without numerical dependencies. The typed trace layer uses Pydantic; the installed package requires its declared dependencies.

    Public API

    • :class:AgentConfig -- configuration dataclass (max_steps, convergence, etc.)
    • :class:AgentStep -- immutable record of one inference step
    • :class:AgentRuntime -- the core runtime wrapping a matrices module
    • :func:run_n_steps -- convenience: run N steps
    • :func:run_until_convergence -- convenience: run until belief stabilises

    @dataclass
class AgentConfig:
    """Configuration for the Active Inference agent loop.

    Attributes:
        max_steps: Maximum number of inference steps before forced halt.
            Must be non-negative.
        convergence_threshold: KL divergence threshold below which the
            agent considers its belief to have converged. Must be in (0, 1).
    """

    max_steps: int = 100
    convergence_threshold: float = 1e-4

    def __post_init__(self) -> None:
        """Validate configuration after initialization.

        Raises:
            ValueError: If any configuration value is invalid.
        """
        if type(self.max_steps) is not int or self.max_steps < 0:
            raise ValueError(f"max_steps must be a non-negative integer; got {self.max_steps}")

        if (
            isinstance(self.convergence_threshold, bool)
            or not isinstance(self.convergence_threshold, (int, float))
            or not math.isfinite(self.convergence_threshold)
            or not (0 < self.convergence_threshold < 1)
        ):
            raise ValueError(
                f"convergence_threshold must be in (0, 1); got {self.convergence_threshold}"
            )

    @classmethod
    def from_yaml(cls, path: str) -> AgentConfig:
        """Load configuration from a YAML file.

        Args:
            path: Path to YAML file with keys matching AgentConfig fields.

        Returns:
            Loaded AgentConfig instance.

        Raises:
            ImportError: If pyyaml is not installed.
            FileNotFoundError: If the file does not exist.
            ValueError: If YAML contains invalid values.

        Example:
            >>> cfg = AgentConfig.from_yaml("config.yaml")
        """
        try:
            import yaml
        except ImportError as exc:
            raise ImportError(
                "pyyaml is required for from_yaml(); install it with: pip install pyyaml"
            ) from exc

        with open(path, encoding="utf-8") as f:
            data = yaml.safe_load(f)
        if data is None:
            data = {}
        if not isinstance(data, dict):
            raise ValueError("runtime configuration must be a YAML mapping")
        unknown = set(data) - {"max_steps", "convergence_threshold"}
        if unknown:
            raise ValueError(f"unknown runtime configuration fields: {sorted(map(str, unknown))}")
        return cls(**data)

    def to_yaml(self, path: str) -> None:
        """Save configuration to a YAML file.

        Args:
            path: Path where YAML file will be written.

        Raises:
            ImportError: If pyyaml is not installed.

        Example:
            >>> cfg = AgentConfig(max_steps=200)
            >>> cfg.to_yaml("config.yaml")
        """
        try:
            import yaml
        except ImportError as exc:
            raise ImportError(
                "pyyaml is required for to_yaml(); install it with: pip install pyyaml"
            ) from exc

        data = {
            "max_steps": self.max_steps,
            "convergence_threshold": self.convergence_threshold,
        }

        with open(path, "w", encoding="utf-8") as f:
            yaml.safe_dump(data, f, default_flow_style=False)

        logger.info(f"Saved AgentConfig to {path}")

    @classmethod
    def with_defaults(cls) -> AgentConfig:
        """Create a new AgentConfig with sensible defaults.

        Returns:
            A new AgentConfig instance with default values.

        Example:
            >>> cfg = AgentConfig.with_defaults()
            >>> print(cfg.max_steps)
            100
        """
        return cls()

    @model_validator(mode="after")

            raise ValueError("observation count must equal the requested steps")

    class InferenceTrace(BaseModel):
    """Versioned execution artifact bound to exact GNN text and parsed matrices."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True)
    schema_version: Literal["2.0"] = "2.0"
    source_sha256: Digest
    matrices_sha256: Digest
    source_artifact: str | None = None
    model_name: str
    mode: Literal["model_rollout", "observations"]
    policy_horizon: Literal[1] = 1
    policy_objective: Literal["risk_plus_ambiguity"] = "risk_plus_ambiguity"
    labels: AxisLabels
    matrices: MatrixSnapshot
    trace: Annotated[tuple[InferenceRecord, ...], Field(min_length=1, max_length=10_000)]

    @model_validator(mode="after")
    def execution_axes(self) -> Self:
        if self.matrices.digest() != self.matrices_sha256:
            raise ValueError("matrix snapshot digest does not match the trace identity")
        ns, no = len(self.matrices.D), len(self.matrices.A)
        na = len(self.matrices.B[0][0])
        if tuple(map(len, (self.labels.states, self.labels.observations, self.labels.actions))) != (
            ns,
            no,
            na,
        ):
            raise ValueError("trace labels do not match the matrix axes")
        if not self.trace:
            raise ValueError("an inference trace must contain at least one observation")
        prior = self.matrices.D
        for t, record in enumerate(self.trace):
            if record.t != t or record.prior != prior:
                raise ValueError("trace timing or incoming prior is inconsistent")
            for vector in (record.prior, record.posterior, record.predicted_belief):
                if len(vector) != ns:
                    raise ValueError("trace belief has incompatible state dimensions")
                distribution(vector, "trace belief")
            if len(record.predicted_observation) != no or record.observation >= no:
                raise ValueError("trace observation has incompatible dimensions")
            distribution(record.predicted_observation, "predicted observation")
            if len(record.policy_scores) != na or record.action >= na:
                raise ValueError("trace policy has incompatible action dimensions")
            prior = record.predicted_belief
        return self

    class AgentRuntime:
    """Mutable agent matrices with shared categorical inference and learning."""

    def __init__(self, matrices: Any) -> None:
        self.A: list[list[float]] = matrices.A
        self.B: list[list[list[float]]] = matrices.B
        self.C: list[float] = matrices.C
        self.D: list[float] = matrices.D
        self._model = CategoricalModel(self.A, self.B, self.C, self.D)
        self._likelihood = self._model.observe
        self._transition = self._model.predict
        self._n_states = len(self.D)
        self._n_obs = len(self.A)
        self._n_actions = self._model.n_actions
        self._episode_count = 0
        self._last_free_energy = float("nan")

    @classmethod
    def from_matrices_dict(cls, d: dict[str, Any]) -> AgentRuntime:
        return cls(types.SimpleNamespace(A=d["A"], B=d["B"], C=d["C"], D=d["D"]))

    def step(self, state_dist: list[float], obs_idx: int, t: int = 0) -> AgentStep:
        """Filter then choose minimum one-step EFE, with lowest-index ties."""
        result = self._model.step(state_dist, obs_idx)
        self._last_free_energy = result.free_energy
        return AgentStep(
            t,
            result.prediction,
            obs_idx,
            result.action,
            result.free_energy,
            result.posterior,
            result.policy_scores,
        )

    def run_n_steps(self, n: int, initial_state: list[float] | None = None) -> list[AgentStep]:
        """Run a deterministic model rollout using modal predicted observations."""
        if n < 0:
            raise ValueError("step count must be non-negative")
        state = list(self.D if initial_state is None else initial_state)
        distribution(state)
        steps = []
        for t in range(n):
            result = self.step(state, _argmax(self._likelihood(state)), t)
            steps.append(result)
            state = result.state_dist
        return steps

    def run_until_convergence(
        self, initial_state: list[float] | None = None, cfg: AgentConfig | None = None
    ) -> list[AgentStep]:
        cfg = cfg if cfg is not None else AgentConfig()
        state = list(self.D if initial_state is None else initial_state)
        distribution(state)
        steps = []
        for t in range(cfg.max_steps):
            result = self.step(state, _argmax(self._likelihood(state)), t)
            steps.append(result)
            if t > 0 and kl_divergence(result.state_dist, state) < cfg.convergence_threshold:
                break
            state = result.state_dist
        return steps

    def run_episode(self, n_steps: int, initial_state: list[float] | None = None) -> EpisodeResult:
        initial = list(self.D if initial_state is None else initial_state)
        return summarize_episode(self.run_n_steps(n_steps, initial), initial, self._n_obs)

    def update_D_from_posterior(self, posterior: list[float]) -> list[float]:
        if len(posterior) != self._n_states:
            raise ValueError("posterior has incompatible state dimensions")
        normalized = _normalize(posterior)
        k = self._episode_count
        self.D[:] = [(d * k + q) / (k + 1) for d, q in zip(self.D, normalized, strict=True)]
        self._episode_count += 1
        return self.D

    def update_A_from_counts(
        self, obs_state_counts: list[list[float]], learning_rate: float = 0.1
    ) -> list[list[float]]:
        """Blend conditional P(observation | state) counts, leaving unseen states unchanged."""
        if not math.isfinite(learning_rate) or not 0 <= learning_rate <= 1:
            raise ValueError("learning_rate must be finite and in [0, 1]")
        if len(obs_state_counts) != self._n_obs or any(
            len(row) != self._n_states for row in obs_state_counts
        ):
            raise ValueError("observation-state counts have incompatible dimensions")
        if any(not math.isfinite(v) or v < 0 for row in obs_state_counts for v in row):
            raise ValueError("observation-state counts must be finite and non-negative")
        for s in range(self._n_states):
            total = math.fsum(row[s] for row in obs_state_counts)
            if total > 0:
                for o in range(self._n_obs):
                    self.A[o][s] = (1 - learning_rate) * self.A[o][
                        s
                    ] + learning_rate * obs_state_counts[o][s] / total
        return self.A

    def run_multi_episode(
        self,
        n_episodes: int,
        steps_per_episode: int,
        learning_rate: float = 0.1,
        initial_state: list[float] | None = None,
    ) -> MultiEpisodeResult:
        if n_episodes < 0:
            raise ValueError("episode count must be non-negative")
        episodes = []
        priors = []
        for _ in range(n_episodes):
            result = self.run_episode(steps_per_episode, initial_state)
            episodes.append(result)
            self.update_D_from_posterior(result.final_posterior)
            self.update_A_from_counts(result.obs_state_counts, learning_rate)
            priors.append(list(self.D))
        return MultiEpisodeResult(
            episodes,
            [e.mean_free_energy for e in episodes],
            [e.final_free_energy for e in episodes],
            priors,
            learning_rate,
        )

    def run_episode_with_logging(
        self, obs_sequence: list[int]
    ) -> tuple[EpisodeResult, list[dict[str, Any]]]:
        state = list(self.D)
        steps = []
        logs = []
        for t, obs in enumerate(obs_sequence):
            result = self.step(state, obs, t)
            steps.append(result)
            logs.append(
                {
                    "step": t,
                    "observation": obs,
                    "action": result.action,
                    "free_energy": result.free_energy,
                }
            )
            state = result.state_dist
        return summarize_episode(steps, self.D, self._n_obs), logs

    def benchmark(self, n_episodes: int = 10) -> dict[str, float]:
        if n_episodes <= 0:
            raise ValueError("benchmark episode count must be positive")
        elapsed = []
        for _ in range(n_episodes):
            start = time.perf_counter()
            self.run_episode(5)
            elapsed.append((time.perf_counter() - start) * 1000)
        mean = sum(elapsed) / len(elapsed)
        return {
            "mean_ms": mean,
            "std_ms": (sum((x - mean) ** 2 for x in elapsed) / len(elapsed)) ** 0.5,
            "min_ms": min(elapsed),
            "max_ms": max(elapsed),
        }

    def reset(self) -> None:
        self._episode_count = 0
        self._last_free_energy = float("nan")

    def get_free_energy(self) -> float:
        return self._last_free_energy

    def to_dict(self) -> dict[str, Any]:
        return {
            "A": self.A,
            "B": self.B,
            "C": self.C,
            "D": self.D,
            "n_states": self._n_states,
            "n_obs": self._n_obs,
            "n_actions": self._n_actions,
            "episode_count": self._episode_count,
        }

    def from_dict(cls, d: dict[str, Any]) -> AgentRuntime:
        runtime = cls.from_matrices_dict(d)
        runtime._episode_count = d.get("episode_count", 0)
        return runtime

    def run_gnn_inference(text: str, options: InferenceOptions | None = None) -> InferenceTrace:
    """Execute parsed GNN matrices without code generation; never invent missing matrices."""
    options = options or InferenceOptions()
    model = parse_gnn(text)
    if model.degraded:
        raise ValueError(
            "inference requires a complete GNN representation without degraded parsing"
        )
    functions = MatrixFunctions(model)
    matrices = MatrixSnapshot(
        A=tuple(tuple(row) for row in functions.A),
        B=tuple(tuple(tuple(cell) for cell in row) for row in functions.B),
        C=tuple(functions.C),
        D=tuple(functions.D),
    )
    ns, no, na = len(matrices.D), len(matrices.A), len(matrices.B[0][0])
    work = options.steps * na * (ns * ns + ns * no)
    if work > options.max_work_units:
        raise ValueError(f"inference work {work} exceeds budget {options.max_work_units}")
    if options.observations is not None and any(o >= no for o in options.observations):
        raise ValueError("observation index is outside the GNN observation axis")
    runtime = AgentRuntime(functions)
    state = list(functions.D)
    records: list[InferenceRecord] = []
    for t in range(options.steps):
        predicted_observation = functions.likelihood(state)
        observation = (
            options.observations[t]
            if options.observations is not None
            else max(range(no), key=predicted_observation.__getitem__)
        )
        step = runtime.step(state, observation, t=t)
        records.append(
            InferenceRecord(
                t=t,
                prior=tuple(state),
                posterior=tuple(step.posterior),
                predicted_belief=tuple(step.state_dist),
                predicted_observation=tuple(predicted_observation),
                observation=observation,
                action=step.action,
                policy_scores=tuple(step.policy_scores),
                free_energy=step.free_energy,
            )
        )
        state = list(step.state_dist)
    return InferenceTrace(
        source_sha256=hashlib.sha256(text.encode("utf-8")).hexdigest(),
        matrices_sha256=matrices.digest(),
        model_name=model.raw_model_name,
        mode="observations" if options.observations is not None else "model_rollout",
        labels=AxisLabels(
            states=_axis_labels(model.hidden_states, ns, "state"),
            observations=_axis_labels(model.observations, no, "observation"),
            actions=_axis_labels(model.actions, na, "action"),
        ),
        matrices=matrices,
        trace=tuple(records),
    )

        state = list(self.D if initial_state is None else initial_state)
        distribution(state)
        steps = []
        for t in range(cfg.max_steps):
            result = self.step(state, _argmax(self._likelihood(state)), t)
            steps.append(result)
            if t > 0 and kl_divergence(result.state_dist, state) < cfg.convergence_threshold:
                break
            state = result.state_dist
        return steps

    def run_episode(self, n_steps: int, initial_state: list[float] | None = None) -> EpisodeResult:
        initial = list(self.D if initial_state is None else initial_state)
        return summarize_episode(self.run_n_steps(n_steps, initial), initial, self._n_obs)

    def update_D_from_posterior(self, posterior: list[float]) -> list[float]:
        if len(posterior) != self._n_states:
            raise ValueError("posterior has incompatible state dimensions")
        normalized = _normalize(posterior)
        k = self._episode_count
        self.D[:] = [(d * k + q) / (k + 1) for d, q in zip(self.D, normalized, strict=True)]
        self._episode_count += 1
        return self.D

    def update_A_from_counts(
        self, obs_state_counts: list[list[float]], learning_rate: float = 0.1
    ) -> list[list[float]]:
        """Blend conditional P(observation | state) counts, leaving unseen states unchanged."""
        if not math.isfinite(learning_rate) or not 0 <= learning_rate <= 1:
            raise ValueError("learning_rate must be finite and in [0, 1]")
        if len(obs_state_counts) != self._n_obs or any(
            len(row) != self._n_states for row in obs_state_counts
        ):
            raise ValueError("observation-state counts have incompatible dimensions")
        if any(not math.isfinite(v) or v < 0 for row in obs_state_counts for v in row):
            raise ValueError("observation-state counts must be finite and non-negative")
        for s in range(self._n_states):
            total = math.fsum(row[s] for row in obs_state_counts)
            if total > 0:
                for o in range(self._n_obs):
                    self.A[o][s] = (1 - learning_rate) * self.A[o][
                        s
                    ] + learning_rate * obs_state_counts[o][s] / total
        return self.A

    def run_multi_episode(
        self,
        n_episodes: int,
        steps_per_episode: int,
        learning_rate: float = 0.1,
        initial_state: list[float] | None = None,
    ) -> MultiEpisodeResult:
        if n_episodes < 0:
            raise ValueError("episode count must be non-negative")
        episodes = []
        priors = []
        for _ in range(n_episodes):
            result = self.run_episode(steps_per_episode, initial_state)
            episodes.append(result)
            self.update_D_from_posterior(result.final_posterior)
            self.update_A_from_counts(result.obs_state_counts, learning_rate)
            priors.append(list(self.D))
        return MultiEpisodeResult(
            episodes,
            [e.mean_free_energy for e in episodes],
            [e.final_free_energy for e in episodes],
            priors,
            learning_rate,
        )

    def run_episode_with_logging(
        self, obs_sequence: list[int]
    ) -> tuple[EpisodeResult, list[dict[str, Any]]]:
        state = list(self.D)
        steps = []
        logs = []
        for t, obs in enumerate(obs_sequence):
            result = self.step(state, obs, t)
            steps.append(result)
            logs.append(
                {
                    "step": t,
                    "observation": obs,
                    "action": result.action,
                    "free_energy": result.free_energy,
                }
            )
            state = result.state_dist
        return summarize_episode(steps, self.D, self._n_obs), logs

    def benchmark(self, n_episodes: int = 10) -> dict[str, float]:
        if n_episodes <= 0:
            raise ValueError("benchmark episode count must be positive")
        elapsed = []
        for _ in range(n_episodes):
            start = time.perf_counter()
            self.run_episode(5)
            elapsed.append((time.perf_counter() - start) * 1000)
        mean = sum(elapsed) / len(elapsed)
        return {
            "mean_ms": mean,
            "std_ms": (sum((x - mean) ** 2 for x in elapsed) / len(elapsed)) ** 0.5,
            "min_ms": min(elapsed),
            "max_ms": max(elapsed),
        }

    def reset(self) -> None:

class AgentConfig:
    """Configuration for the Active Inference agent loop.

    Attributes:
        max_steps: Maximum number of inference steps before forced halt.
            Must be non-negative.
        convergence_threshold: KL divergence threshold below which the

    convergence_threshold: float = 1e-4

    def __post_init__(self) -> None:
        """Validate configuration after initialization.

        Raises:
            ValueError: If any configuration value is invalid.
        """
        if type(self.max_steps) is not int or self.max_steps < 0:
            raise ValueError(f"max_steps must be a non-negative integer; got {self.max_steps}")

        if (
            isinstance(self.convergence_threshold, bool)
            or not isinstance(self.convergence_threshold, (int, float))
            or not math.isfinite(self.convergence_threshold)
            or not (0 < self.convergence_threshold < 1)
        ):
            raise ValueError(
                f"convergence_threshold must be in (0, 1); got {self.convergence_threshold}"
            )

        """Load configuration from a YAML file.

            ImportError: If pyyaml is not installed.

            import yaml

    """Aggregate metrics for a multi-episode learning run.

    Attributes:
        episodes: List of per-episode metrics.
        total_steps: Total inference steps across all episodes.
    """

    episodes: list[EpisodeMetrics]
    total_steps: int = 0

    def summary_statistics(self) -> dict[str, float]:
        """Compute summary statistics over all episodes.

        Returns:
            Dictionary with mean/std/min/max for:
            * ``mean_free_energy_mean``, ``mean_free_energy_std``, etc.
            * ``final_free_energy_mean``, ``final_free_energy_std``, etc.

        Example:
            >>> metrics = RunMetrics(episodes=[...])
            >>> stats = metrics.summary_statistics()
            >>> print(f"Mean VFE: {stats['mean_free_energy_mean']:.4f}")
        """
        if not self.episodes:
            logger.warning("No episodes in RunMetrics; returning NaN-filled stats")
            nan = float("nan")
            return {
                "n_episodes": 0,
                "total_steps": 0,
                "mean_free_energy_mean": nan,
                "mean_free_energy_std": nan,
                "mean_free_energy_min": nan,
                "mean_free_energy_max": nan,
                "final_free_energy_mean": nan,
                "final_free_energy_std": nan,
                "final_free_energy_min": nan,
                "final_free_energy_max": nan,
                "steps_per_episode_mean": nan,
                "steps_per_episode_std": nan,
            }

        mean_fes = [e.mean_free_energy for e in self.episodes]
        final_fes = [e.final_free_energy for e in self.episodes]
        step_counts = [e.n_steps for e in self.episodes]

        def _stats(values: list[float]) -> tuple[float, float, float, float]:
            """Compute mean, std, min, max."""
            if not values:
                return 0.0, 0.0, 0.0, 0.0
            mean = sum(values) / len(values)
            variance = sum((v - mean) ** 2 for v in values) / len(values)
            std = variance**0.5
            return mean, std, min(values), max(values)

    final_fes = [e.final_free_energy for e in self.episodes]

    def _stats(values: list[float]) -> tuple[float, float, float, float]:
        """Compute mean, std, min, max."""
        if not values:
            return 0.0, 0.0, 0.0, 0.0
        mean = sum(values) / len(values)
        variance = sum((v - mean) ** 2 for v in values) / len(values)
        std = variance**0.5
        return mean, std, min(values), max(values)

    mean_fe_mean, mean_fe_std, mean_fe_min, mean_fe_max = _stats(mean_fes)
    final_fe_mean, final_fe_std, final_fe_min, final_fe_max = _stats(final_fes)
    step_mean, step_std, step_min, step_max = _stats([float(s) for s in step_counts])

    return {
        "n_episodes": len(self.episodes),
        "total_steps": sum(step_counts),
        "mean_free_energy_mean": mean_fe_mean,
        "mean_free_energy_std": mean_fe_std,
